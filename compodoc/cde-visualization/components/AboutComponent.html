import { ChangeDetectionStrategy, Component, output } from '@angular/core';
import { MatIconModule } from '@angular/material/icon';
import { HraCommonModule } from '@hra-ui/common';
import { ButtonsModule } from '@hra-ui/design-system/buttons';
import { ScrollingModule } from '@hra-ui/design-system/scrolling';
import { PlainTooltipDirective } from '@hra-ui/design-system/tooltips/plain-tooltip';
import { ABOUT_CONTENT } from './about.content';

/** Counter to keep track of distinct instances */
let idCounter = 0;

/**
 * Describes each panel of the visualization. Rendered inside the drawer when the toolbar's Info item is clicked.
 */
@Component({
  selector: 'cde-about',
  imports: [HraCommonModule, MatIconModule, ButtonsModule, PlainTooltipDirective, ScrollingModule],
  templateUrl: './about.component.html',
  styleUrl: './about.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class AboutComponent {
  /** Emits when the close button is clicked */
  readonly closeClick = output<void>();

  /** Id attribute for the title, used to label the drawer */
  readonly titleId = `cde-about-title-${idCounter}`;

  /** Id attribute for the description, used to describe the drawer */
  readonly descriptionId = `cde-about-description-${idCounter++}`;

  /** Text content */
  protected readonly content = ABOUT_CONTENT;
}

<div class="header">
  <h2 class="title" [id]="titleId">{{ content.title }}</h2>
  <button
    mat-icon-button
    class="close"
    hraFeature="close"
    hraClickEvent
    [attr.aria-label]="content.closeLabel"
    [hraPlainTooltip]="content.closeLabel"
    (click)="closeClick.emit()"
  >
    <mat-icon>close</mat-icon>
  </button>
  <p class="description" [id]="descriptionId">{{ content.description }}</p>
</div>

<ng-scrollbar class="body" orientation="vertical" hraScrollOverflowFade>
  <div class="sections">
    @for (section of content.sections; track section.heading) {
      <section class="section">
        <h3 class="section-heading">{{ section.heading }}</h3>
        <p class="section-description">{{ section.description }}</p>
        @for (list of section.lists; track list.heading) {
          <h4 class="list-heading">{{ list.heading }}</h4>
          <ul class="list">
            @for (item of list.items; track item) {
              <li>{{ item }}</li>
            }
          </ul>
        }
      </section>
    }
  </div>
</ng-scrollbar>
