import { ChangeDetectionStrategy, Component, input } from '@angular/core';
import { MatRippleModule } from '@angular/material/core';
import { MatIconModule } from '@angular/material/icon';
import { HraCommonModule } from '@hra-ui/common';
import { LinkDirective } from '@hra-ui/common/router-ext';

/** Example dataset shown in the list */
export interface ExampleItem {
  /** Example name */
  title: string;
  /** Short metadata line, such as technology and cell count */
  description: string;
  /** Credit for the lab that produced the data */
  credit?: string;
  /** Thumbnail image url, 16:10 */
  image: string;
  /** Route that opens the example */
  url: string;
}

/**
 * Compact list of example datasets where each row links to its example.
 */
@Component({
  selector: 'cde-example-list',
  imports: [HraCommonModule, LinkDirective, MatIconModule, MatRippleModule],
  templateUrl: './example-list.component.html',
  styleUrl: './example-list.component.scss',
  changeDetection: ChangeDetectionStrategy.OnPush,
})
export class ExampleListComponent {
  /** Examples to list */
  readonly examples = input.required<ExampleItem[]>();
}

<ul class="examples">
  @for (example of examples(); track example.url) {
    <li>
      <a class="example" hraFeature="example" matRipple [hraClickEvent]="{ url: example.url }" [hraLink]="example.url">
        <span class="thumbnail">
          <img alt="" [attr.src]="example.image | assetUrl" />
        </span>
        <span class="text">
          <span class="title">{{ example.title }}</span>
          <span class="description">{{ example.description }}</span>
          @if (example.credit) {
            <span class="credit">{{ example.credit }}</span>
          }
        </span>
        <span class="arrow">
          <mat-icon>arrow_right_alt</mat-icon>
        </span>
      </a>
    </li>
  }
</ul>
